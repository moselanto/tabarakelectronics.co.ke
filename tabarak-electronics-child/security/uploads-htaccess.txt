# ============================================================
# Anti-malware guard for the uploads folder.
# Save as: .../wp-content/uploads/.htaccess
# Stops PHP/scripts from RUNNING inside uploads (the #1 way an
# attacker executes malware after sneaking a file in). Images/PDFs still work.
# ============================================================

<FilesMatch "(?i)\.(php|php3|php4|php5|php7|php8|phtml|pht|phps|phar|cgi|pl|py|asp|aspx|jsp|sh|bash|exe|so)$">
  Require all denied
</FilesMatch>
Options -Indexes
